def lock_profile(
    db: Session,
    profile_id: UUID,
) -> Profile | None:
    """
    Bloqueia a linha do perfil durante a geraÃ§Ã£o de looks.

    SELECT FOR UPDATE evita que duas geraÃ§Ãµes simultÃ¢neas
    do mesmo perfil registrem a mesma combinaÃ§Ã£o.
    """

    statement = select(Profile).where(Profile.id == profile_id).with_for_update()

    return db.scalar(statement)


def delete_expired(
    db: Session,
    profile_id: UUID,
    now: datetime,
) -> int:
    """
    Remove looks vencidos que nÃ£o foram favoritados.

    Looks favoritados sÃ£o preservados independentemente
    de expires_at, pois representam uma escolha explÃ­cita
    do usuÃ¡rio.

    O CASCADE do banco remove os OutfitItems associados.
    """

    statement = (
        delete(Outfit)
        .where(
            Outfit.profile_id == profile_id,
            Outfit.expires_at <= now,
            Outfit.is_favorite.is_(False),
        )
        .returning(Outfit.id)
    )

    return len(db.scalars(statement).all())


def list_available_garments(
    db: Session,
    profile_id: UUID,
) -> list[Garment]:
    """
    Busca somente peÃ§as disponÃ­veis pertencentes ao perfil.

    A filtragem por profile_id Ã© uma regra de seguranÃ§a:
    nunca buscamos peÃ§as de outros usuÃ¡rios.
    """

    statement = (
        select(Garment)
        .where(
            Garment.profile_id == profile_id,
            Garment.status == "available",
        )
        .order_by(Garment.created_at.asc())
    )

    return list(db.scalars(statement).all())


def list_recent_signatures(
    db: Session,
    profile_id: UUID,
    now: datetime,
) -> set[str]:
    """
    Retorna as assinaturas que nÃ£o podem ser reutilizadas.

    Considera:
    - looks que ainda estÃ£o dentro do prazo de retenÃ§Ã£o;
    - looks favoritados, mesmo que jÃ¡ tenham vencido.

    statement = select(Outfit.signature).where(
        Outfit.profile_id == profile_id,
        ((Outfit.expires_at > now) | Outfit.is_favorite.is_(True)),

    return set(db.scalars(statement).all())


def save_outfit(
    db: Session,
    outfit: Outfit,
    garment_roles: list[tuple[Garment, str]],
) -> Outfit:
    """
    Persiste o look e suas peÃ§as em uma Ãºnica transaÃ§Ã£o.

    Se alguma associaÃ§Ã£o falhar, a transaÃ§Ã£o poderÃ¡ ser
    revertida sem deixar um look parcialmente cadastrado.
    """

    db.add(outfit)
    db.flush()

    for garment, role in garment_roles:
        db.add(
            OutfitItem(
                outfit_id=outfit.id,
                garment_id=garment.id,
                role=role,
            )
        )

    db.flush()

    return outfit

def generate_outfit(

    """

                used_signatures=used_signatures,

            db.commit()

        outfit = Outfit(
            profile_id=profile_id,

            season=request.season,
            formality=request.formality,
            signature=signature,
            generated_at=now,

            db=db,
            outfit=outfit,

            profile_id=profile_id,

from hashlib import sha256
from random import SystemRandom
from uuid import UUID

from app.models.garment import Garment

# Quantidade mÃ¡xima de tentativas para encontrar uma combinaÃ§Ã£o
# que ainda nÃ£o esteja no histÃ³rico.
MAX_GENERATION_ATTEMPTS = 300

# Categorias que podem complementar o look.
# Cada uma pode contribuir com, no mÃ¡ximo, uma peÃ§a.
OPTIONAL_CATEGORIES = {
    "bags": "bag",
    "accessories": "accessory",
    "jewelry": "jewelry",
    "hair_accessories": "hair_accessory",
}


class OutfitCombinationError(Exception):
    """NÃ£o foi possÃ­vel montar uma combinaÃ§Ã£o vÃ¡lida."""


def create_signature(garment_ids: list[UUID]) -> str:
    """
    Cria uma assinatura estÃ¡vel para uma combinaÃ§Ã£o.

    Ordenar os IDs antes de calcular o hash garante que
    a mesma combinaÃ§Ã£o tenha a mesma assinatura,
    independentemente da ordem das peÃ§as.
    """

    normalized_ids = sorted(str(garment_id) for garment_id in garment_ids)

    content = "|".join(normalized_ids)

    return sha256(content.encode("utf-8")).hexdigest()


def matches_context(
    garment: Garment,
    occasion: str | None,
    season: str | None,
    formality: int | None,
) -> bool:
    """
    Verifica se uma peÃ§a atende aos filtros solicitados.

    Uma peÃ§a sem informaÃ§Ã£o de ocasiÃ£o ou estaÃ§Ã£o continua
    elegÃ­vel, porque nÃ£o temos dados suficientes para descartÃ¡-la.

    A formalidade aceita uma diferenÃ§a de atÃ© um nÃ­vel.
    """

    if occasion:
        allowed_occasions = {value.casefold() for value in garment.occasion_tags}

        if allowed_occasions and occasion.casefold() not in allowed_occasions:
            return False

    if season:
        allowed_seasons = {value.casefold() for value in garment.season_tags}

        if allowed_seasons and season.casefold() not in allowed_seasons:
            return False

    if formality is not None and garment.formality is not None:
        if abs(garment.formality - formality) > 1:
            return False

    return True


def generate_candidate(
    garments: list[Garment],
    occasion: str | None,
    season: str | None,
    formality: int | None,
    rng: SystemRandom,
) -> list[tuple[Garment, str]]:
    """
    Monta uma combinaÃ§Ã£o candidata.

    Estruturas permitidas:
    - top + bottom + shoes
    - one_piece + shoes

    Complementos sÃ£o opcionais.
    """

    eligible = [
        garment
        for garment in garments
        if matches_context(
            garment=garment,
            occasion=occasion,
            season=season,
            formality=formality,
        )
    ]

    by_category: dict[str, list[Garment]] = {}

    for garment in eligible:
        by_category.setdefault(garment.category, []).append(garment)

    shoes = by_category.get("shoes", [])

    # Sem calÃ§ados nÃ£o geramos uma combinaÃ§Ã£o completa.
    if not shoes:
        raise OutfitCombinationError("Cadastre pelo menos um calÃ§ado disponÃ­vel para gerar looks.")

    # Existem duas estruturas possÃ­veis para o look.
    structures = []

    if by_category.get("tops") and by_category.get("bottoms"):
        structures.append("separates")

    if by_category.get("one_piece"):
        structures.append("one_piece")

    if not structures:
        raise OutfitCombinationError(
            "Seu closet precisa ter uma parte de cima e uma parte "
            "de baixo, ou uma peÃ§a Ãºnica, alÃ©m de um calÃ§ado."
        )

    selected_structure = rng.choice(structures)

    selected: list[tuple[Garment, str]] = []

    if selected_structure == "separates":
        selected.append((rng.choice(by_category["tops"]), "top"))
        selected.append((rng.choice(by_category["bottoms"]), "bottom"))
    else:
        selected.append((rng.choice(by_category["one_piece"]), "one_piece"))

    selected.append((rng.choice(shoes), "shoes"))

    # Complementos nÃ£o sÃ£o obrigatÃ³rios.
    # Se houver peÃ§as disponÃ­veis, o algoritmo decide
    # aleatoriamente se cada categoria serÃ¡ utilizada.
    for category, role in OPTIONAL_CATEGORIES.items():
        options = by_category.get(category, [])

        if options and rng.choice((True, False)):
            selected.append((rng.choice(options), role))

    return selected


def generate_unique_combination(
    garments: list[Garment],
    used_signatures: set[str],
    occasion: str | None,
    season: str | None,
    formality: int | None,
) -> tuple[list[tuple[Garment, str]], str]:
    """
    Procura uma combinaÃ§Ã£o que ainda nÃ£o tenha sido utilizada.

    O algoritmo nÃ£o recebe IDs de peÃ§as enviados pelo cliente.
    Todas as peÃ§as sÃ£o selecionadas a partir do closet autenticado.
    """

    rng = SystemRandom()

    # Verifica previamente se existe alguma estrutura bÃ¡sica possÃ­vel.
    eligible = [
        garment
        for garment in garments
        if matches_context(
            garment,
            occasion,
            season,
            formality,
        )
    ]

    categories = {garment.category for garment in eligible}

    has_separates = {"tops", "bottoms", "shoes"}.issubset(categories)
    has_one_piece = {"one_piece", "shoes"}.issubset(categories)

    if not has_separates and not has_one_piece:
        raise OutfitCombinationError(
            "NÃ£o existem peÃ§as suficientes e compatÃ­veis para gerar um look."
        )

    for _ in range(MAX_GENERATION_ATTEMPTS):
        candidate = generate_candidate(
            garments=garments,
            occasion=occasion,
            season=season,
            formality=formality,
            rng=rng,
        )

        signature = create_signature([garment.id for garment, _ in candidate])

        if signature not in used_signatures:
            return candidate, signature

    raise OutfitCombinationError(
        "NÃ£o encontrei uma combinaÃ§Ã£o inÃ©dita com as peÃ§as disponÃ­veis. "
        "Adicione novas peÃ§as ao closet ou tente novamente apÃ³s o perÃ­odo "
        "de validade dos looks anteriores."
    )
